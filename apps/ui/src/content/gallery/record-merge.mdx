The formatted card sets `formatValue`:

```ts tag=CodeBlock
merge.formatValue = ({ field, value }) => {
  if (value === undefined || value === null || value === '') return html`<span class="merge-none">none</span>`
  if (field.key === 'created') return new Date(String(value)).toLocaleDateString('en-US', { dateStyle: 'medium' })
  if (field.key === 'revenue') return money.format(Number(value))
  return String(value)
}
```
