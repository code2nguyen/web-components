## Presets

Looks and cases for `c2-schema-tree`. The `--c2-schema-tree__*` variables colour the type badge, the marks, the
constraints and the description; the `--c2-json-viewer-*` variables still set the box, the rows and the names. Open any
card in the studio to start from it.

<GalleryGrid>

```html tag=MdxCodeBlock,label=Default
<c2-schema-tree
  schema='{"type":"object","required":["name"],"properties":{"name":{"type":"string","maxLength":40},"age":{"type":"integer","minimum":0},"admin":{"type":"boolean","default":false},"roles":{"type":"array","items":{"type":"string","enum":["admin","editor","viewer"]}}}}'
></c2-schema-tree>
```

```html tag=MdxCodeBlock,label=References and recursion
<c2-schema-tree
  label="Category"
  schema='{"$ref":"#/$defs/Category","$defs":{"Node":{"type":"object","required":["id"],"properties":{"id":{"type":"string","format":"uuid"},"createdAt":{"type":"string","format":"date-time","readOnly":true}}},"Category":{"allOf":[{"$ref":"#/$defs/Node"},{"properties":{"name":{"type":"string","minLength":1},"parent":{"$ref":"#/$defs/Category"},"children":{"type":"array","items":{"$ref":"#/$defs/Category"}}}}]}}}'
></c2-schema-tree>
```

```html tag=MdxCodeBlock,label=One of
<c2-schema-tree
  label="Payment"
  expand-depth="3"
  schema='{"type":"object","required":["amount","method"],"properties":{"amount":{"type":"number","exclusiveMinimum":0},"method":{"oneOf":[{"title":"Card","type":"object","required":["token"],"properties":{"token":{"type":"string"},"save":{"type":"boolean","default":false}}},{"title":"Bank transfer","type":"object","properties":{"iban":{"type":"string","pattern":"^[A-Z]{2}[0-9]{2}"}}},{"type":"object","properties":{"wallet":{"const":"paypal"}}}]}}}'
></c2-schema-tree>
```

```html tag=MdxCodeBlock,label=OpenAPI component
<c2-schema-tree
  label="Pet"
  pointer="/components/schemas/Pet"
  schema='{"openapi":"3.0.3","components":{"schemas":{"Pet":{"type":"object","required":["name"],"properties":{"id":{"type":"integer","format":"int64","readOnly":true},"name":{"type":"string","description":"What the pet answers to."},"tag":{"type":"string","nullable":true,"deprecated":true,"description":"Use tags instead."},"tags":{"type":"array","uniqueItems":true,"maxItems":5,"items":{"type":"string"}},"owner":{"$ref":"#/components/schemas/Owner"}}},"Owner":{"type":"object","properties":{"email":{"type":"string","format":"email","writeOnly":true}}}}}}'
></c2-schema-tree>
```


The marks and constraints are messages of the component; an app writes them in its own languages:

```ts tag=CodeBlock
import { addTranslations } from '@c2n/core/localize.js'

addTranslations({
  fr: {
    'json-viewer': {
      schema: 'Schéma',
      required: 'obligatoire',
      deprecated: 'obsolète',
      oneOf: 'un parmi',
      option: 'Option {0}',
      maxLength: '≤ {0} caractères',
      default: 'par défaut {0}',
    },
  },
})
```
