The Dialog, Local files and Custom viewer cards are set up from script:

```ts tag=CodeBlock
import type { FilePreview } from '@c2n/components/attachment/file-preview'

const preview = document.querySelector<FilePreview>('c2-file-preview')!
preview.items = [{ name: 'notes.md', file: new File([markdown], 'notes.md', { type: 'text/markdown' }) }]
preview.renderViewer = ({ item }) => (item.name.endsWith('.csv') ? csvTable(item.text ?? '') : undefined)
openButton.addEventListener('click', () => dialogPreview.showModal())
```
