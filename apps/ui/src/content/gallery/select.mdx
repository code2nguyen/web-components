The status, label and server cards are wired by the page, as an application would: the select fires events and never adds, removes or fetches an option itself.

```ts tag=CodeBlock
// Footer actions: the app's own buttons.
clearButton.addEventListener('click', () => (select.value = []))
applyButton.addEventListener('click', () => (select.open = false))

// "Create": add the option and select it.
select.addEventListener('create-option', (event) => {
  const item = document.createElement('c2-list-item')
  item.value = slugify(event.detail.label)
  item.textContent = event.detail.label
  select.append(item)
  select.value = [item.value]
})

// Server search: cancel the built-in filter, show the loading state, replace the options.
select.addEventListener('search-change', async (event) => {
  event.preventDefault()
  select.loading = true
  const people = await searchPeople(event.detail.query)
  select.replaceChildren(...people.map(toListItem))
  select.loading = false
})
```
