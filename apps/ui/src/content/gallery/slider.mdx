## Markers

`markers` draws points of interest on the track: the events of a replay, the chapters of a recording, the readings of a log. They are decoration: the slider still selects any value on its step grid, and a thumb resting on a marker adds its `label` to what assistive technology reads. Each marker's `kind` becomes a `marker-<kind>` part, so a page colours each kind with `::part()`.

The two property-only parts, the time format and the marker content, and the transport around the scrubber are the application's code. Play, pause and speed are a timer that moves `value`; Previous and Next call `seekMarker()`:

```ts tag=CodeBlock
import type { Slider } from '@c2n/components/slider'
import type { Button } from '@c2n/components/button'

const scrubber = document.querySelector('.scrubber')!
const slider = scrubber.querySelector<Slider>('c2-slider')!
const button = (action: string) => scrubber.querySelector<Button>(`[data-action="${action}"]`)!

const clock = (seconds: number) => `${Math.floor(seconds / 60)}:${String(seconds % 60).padStart(2, '0')}`
slider.formatValue = clock

// Play is a toggle button driving a timer that moves `value`; the speed button changes the step.
let speed = 1
let timer: ReturnType<typeof setInterval> | undefined
const stop = () => {
  clearInterval(timer)
  timer = undefined
  button('play').selected = false
}
button('play').addEventListener('click', () => {
  if (timer) return stop()
  if (slider.value >= slider.max) slider.value = slider.min
  timer = setInterval(() => {
    slider.value = Math.min(slider.max, slider.value + speed)
    if (slider.value >= slider.max) stop()
  }, 250)
  button('play').selected = true
})
button('speed').addEventListener('click', () => {
  speed = speed === 4 ? 1 : speed * 2
  button('speed').textContent = `${speed}×`
  button('speed').setAttribute('aria-label', `Speed ${speed}×`)
})

// Previous and Next jump between the markers.
button('previous').addEventListener('click', () => slider.seekMarker(-1))
button('next').addEventListener('click', () => slider.seekMarker(1))

// Marker content: the chapter number inside each dot.
document.querySelector<Slider>('.chapters')!.renderMarker = ({ index }) => String(index + 1)
```
