<HistogramExamplesRuntime />

The French card's text comes from translations the app adds; the live card's samples from a data source whose first
window arrives after a short delay and which then pushes one request every half second:

```ts tag=CodeBlock
import { addTranslations } from '@c2n/core/localize.js'

addTranslations({
  fr: {
    'histogram-chart': {
      bin: 'Classe',
      range: 'de {0} à {1}',
      summary: 'Histogramme de {0}, {1} classes',
      summaryPercentiles: '{0}. {1} : {2}',
      percentileValue: '{0} : {1}',
    },
  },
})

chart.dataSource = {
  getWindow: async () => ({ data: await fetchLatencies() }),
  // Each pushed point adds one sample per series; `max-points` keeps the latest 400.
  subscribe: (push) => onRequest((ms) => push(Date.now(), [ms])),
}
```
