The occurrence renderer card draws each date through `renderOccurrence`, which a fence cannot set:

```ts tag=CodeBlock
import type { RecurrenceEditor } from '@c2n/components/recurrence-editor'
import { html } from 'lit'

const editor = document.querySelector<RecurrenceEditor>('c2-recurrence-editor')!
editor.renderOccurrence = ({ date, label }) => {
  const days = Math.round((date.getTime() - Date.parse('2026-11-16T00:00')) / 86_400_000)
  return html`${label} <strong>${days === 0 ? 'start' : `+${days}d`}</strong>`
}
```

The French card is the editor under `lang="fr"`, with the French an app adds (weekday and month names come from `Intl`):

```ts tag=CodeBlock
addTranslations({

    'recurrence-editor': {
      repeatLabel: 'Répéter',
      everyLabel: 'Tous les',
      summaryMonthly: { one: 'Tous les mois', other: 'Tous les {0} mois' },
      summaryOnNth: 'le {0} {1}',
      ordinal: { '1': 'premier', '2': 'deuxième', '3': 'troisième', '4': 'quatrième', '-1': 'dernier' },
      summaryCount: { one: 'une fois', other: '{0} fois' },
      // …every other message of the API tab
    },
  },
})
```
